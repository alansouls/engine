# matching your current build.zig setup

cmake_minimum_required(VERSION 3.20)
project(Pong VERSION 1.0.0 LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Define preprocessor macros
add_compile_definitions(
    GLFW_INCLUDE_VULKAN 
    GLFW_INCLUDE_NONE
    WINDOWS
)

# Add compiler flags
if(MSVC)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /W4")
else()
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra")
endif()

# Define source files
set(PONG_SOURCES
    Pong/main.cpp
    Pong/game/PongGame.cpp
    Pong/game/gameObjects/Ball.cpp
    Pong/game/gameObjects/Racket.cpp
    Pong/engine/collisions/CollisionManager.cpp
    Pong/engine/collisions/CircleCollider.cpp
    Pong/engine/collisions/QuadCollider.cpp
    Pong/engine/scenes/Scene.cpp
    Pong/engine/scenes/GameObject.cpp
    Pong/engine/scenes/Game.cpp
    Pong/engine/graphics/renderers/CircleItem.cpp
    Pong/engine/graphics/renderers/Renderer.cpp
    Pong/engine/graphics/renderers/RectangleItem.cpp
    Pong/engine/graphics/drivers/VulkanDriver.cpp
)

# Create executable
add_executable(pong ${PONG_SOURCES})

# Set target properties
target_compile_features(pong PRIVATE cxx_std_20)

# Include directories (Windows-specific paths from your build.zig)
target_include_directories(pong PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Pong
    "C:/VulkanSDK/1.3.290.0/Include"
    "C:/Users/maiaa/Documents/Visual Studio 2022/Libraries/glfw-3.4.bin.WIN64/include"
    "C:/Users/maiaa/Documents/Visual Studio 2022/Libraries/glm"
)

# Library directories (Windows-specific paths from your build.zig)
target_link_directories(pong PRIVATE
    "C:/VulkanSDK/1.3.290.0/Lib"
    "C:/Users/maiaa/Documents/Visual Studio 2022/Libraries/glfw-3.4.bin.WIN64/lib-vc2022"
)

# Link libraries
target_link_libraries(pong PRIVATE
    glfw3
    vulkan-1
)

# Copy shader files to build directory
file(GLOB_RECURSE SHADER_FILES 
    "Pong/engine/graphics/drivers/shaders/compiled/*.spv"
)

foreach(SHADER_FILE ${SHADER_FILES})
    get_filename_component(SHADER_NAME ${SHADER_FILE} NAME)
    configure_file(${SHADER_FILE} ${CMAKE_BINARY_DIR}/${SHADER_NAME} COPYONLY)
endforeach()

# Set output directory
set_target_properties(pong PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}/bin
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/bin
)

# Install target
install(TARGETS pong
    RUNTIME DESTINATION bin
)

# Install shaders
install(FILES ${SHADER_FILES}
    DESTINATION bin
)
